#Generating m-combinations of a set of n elements with repetition allowed (in non-decreasing order)(with commas)
k = 0
n = int(input("Enter the value of n: "))
m = int(input("\nEnter the length of the sequence m: "))
print("\n")

def generate(sequence, minimum):
	global k
	if len(sequence) == m:
		print(*sequence, sep = " ,")
		k +=1
		return
		
	for i in range(minimum, n +1):
		generate(sequence + [i], i)
		
generate( [], 1)
print(f"\nThe number of {m}-combinations of a set of {n} elements with repetition allowed is ",k)
